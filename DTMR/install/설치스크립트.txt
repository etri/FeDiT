# 대상: Ubuntu 22.04 이상 (빈 서버)
# 상세 설명은 설치가이드.md 참조
#
# [이 문서를 여는 두 가지 경우]
#   1) 문서만 따로 전달받았다면  -> 0 장부터 순서대로 진행한다.
#                                  0-2 에서 값을 정하고 0-3 에서 소스를 받는다.
#   2) 이미 소스를 받아 그 안의 문서를 보고 있다면
#                                -> 0-3(소스 받기)은 이미 한 것이므로
#                                   0-2 만 실행하되, SRC_DIR 을 소스를 받은
#                                   실제 경로로 맞춘 뒤 1 장으로 넘어간다.
#                                   확인 : ls $SRC_DIR/install
#
# [서버 사양]
#            최소            권장
#   CPU      4 core          8 core
#   RAM      16 GB           32 GB
#   디스크    200 GB SSD      500 GB 이상 (데이터 볼륨 분리)
#
#   한 서버에 저장소 5종(PostgreSQL/Virtuoso/Neo4j/Elasticsearch/InfluxDB)과
#   앱 4종이 함께 뜬다. 기본 설정으로 두면 각자 메모리를 크게 잡으므로
#   운영 서버에서는 아래처럼 힙을 제한한다.
#     Elasticsearch  ES_JAVA_OPTS=-Xms4g -Xmx4g
#     Neo4j          server.memory.heap.max_size=2g / pagecache.size=2g
#     Java API 2종   JAVA_OPTS=-Xmx1g
#
#   디스크는 시간이 지날수록 늘어난다. 아래 두 가지를 설치 시 정해둔다.
#     컨테이너 로그 제한 : docker run 에 --log-opt max-size=100m --log-opt max-file=3
#     InfluxDB 보존기간  : .env.influxdb2 에 DOCKER_INFLUXDB_INIT_RETENTION=365d
################################################################################

########################################
# 0-1. 클라우드 서버인 경우 추가 확인
#      클라우드는 서버 안 방화벽과 별개로 인스턴스 바깥의 방화벽이 하나 더 있다.
#      (AWS 보안그룹 / Azure NSG / NCP ACG)
#      아래 내용은 클라우드 콘솔에서 설정하며, 서버 안에서는 확인만 한다.
#
#      [1] 콘솔에서 열어야 할 인바운드 포트
#          22                     관리자 IP 만
#          8084                   웹 (사용자 접근 범위)
#          33000 / 6998 / 16997   인증·API (브라우저가 직접 호출하므로 외부 개방 필요)
#          7474 / 8890 / 5601 / 8086   관리 콘솔, 관리자 IP 만
#          5432                   DB. 인터넷에 열지 않는다. 필요시 관리자 IP 만
#
#      [2] pg_hba.conf 에 넣을 대역이 달라진다
#          같은 VPC 안에서 접속  -> VPC 사설 대역 (예: 10.0.0.0/16, 172.31.0.0/16)
#          사무실에서 접속       -> 사무실 회선의 공인 IP
#          192.168 대역은 클라우드에서 거의 쓰이지 않는다
#
#      [3] 도커 네트워크 대역이 VPC 대역과 겹치면 안 된다
#          겹치면 컨테이너에서 같은 대역의 다른 서버로 나가지 못한다 (1-5 에서 확인)
#
#      [4] 앱 설정에는 브라우저가 접근하는 주소(공인 IP 또는 도메인)를 넣는다
#          인증서버 serverip / default_url / redirect_uri 에 사설 IP 를 넣으면
#          서버끼리는 통신되지만 로그인 리다이렉트가 깨진다 (7장)
########################################

# 서버 자신의 사설 IP 와 대역 확인 (pg_hba.conf 에 넣을 값)
ip -4 addr show | grep inet

# 이 서버의 공인 IP 확인 (앱 설정에 넣을 값 / 없으면 사설 전용 서버)
curl -s ifconfig.me; echo

# VPC 대역과 도커 대역(172.25) 이 겹치는지 확인
ip route | grep 172.25 || echo "172.25 대역 사용 안 함"


########################################
# 0-2. 설치에 사용할 값 지정
#      여기서 한 번만 정하면 이후 모든 명령(도커 실행, DB 생성, 앱 실행)이 이 값을 쓴다.
#      기본값 그대로 설치해도 되고, 바꿀 값만 고쳐서 실행한다.
#      재접속하면 사라지므로 아래처럼 파일로 만들어 두고 매번 source 한다.
########################################

# 값 파일 생성 (SERVER_HOST 는 실제 서버 IP 로 바꿀 것)
cat > ~/dt.env << 'EOF'

export DT_DIR=$HOME/dtmeta

# 브라우저가 접속하는 주소. 기본은 서버의 첫 번째 IP 를 자동으로 쓴다.
# 사내망에서 접속하면 그대로 두고, 아래에 해당하면 주석을 풀어 직접 지정한다.
#   - 공인 IP 나 도메인으로 접속하는 경우 (클라우드 등)
#   - 서버에 IP 가 여러 개라 다른 IP 로 접속하는 경우
export SERVER_HOST=$(hostname -I | awk '{print $1}')
# export SERVER_HOST=203.0.113.10

export WEB_SERVER_URL=http://$SERVER_HOST:8084
# export WEB_SERVER_URL=https://dtmeta.example.com

# 0-3 에서 받을 소스 경로 (SQL·플러그인·앱 소스가 이 아래에 있다)
# 소스를 다른 경로에 두었다면 이 값을 그 경로로 바꾼다
export SRC_DIR=$DT_DIR/FeDiT/DTMR

# 컴포넌트별 계정명
export DB_USERNAME=meta_manager             # PostgreSQL 메타데이터 DB
export HUBSECURITY_USERNAME=hubsecurity_manager   # PostgreSQL 인증 DB
export VIRTUOSO_USERNAME=meta_manager       # Virtuoso (앱 접속)
export NEO4J_USERNAME=neo4j                 # Neo4j
export ELASTIC_USERNAME=elastic             # Elasticsearch / Kibana
export INFLUX_USERNAME=meta_manager         # InfluxDB

# 컴포넌트별 비밀번호 (같은 값을 써도 되고 각각 다르게 해도 된다)
export DB_PASSWORD='rhksflwk1!'             # PostgreSQL meta_manager / hubsecurity_manager
export VIRTUOSO_DBA_PASSWORD='rhksflwk1!'   # Virtuoso 관리 콘솔(dba)
export VIRTUOSO_PASSWORD='rhksflwk1!'       # Virtuoso meta_manager (앱 접속)
export NEO4J_PASSWORD='rhksflwk1!'          # Neo4j
export ELASTIC_PASSWORD='rhksflwk1!'        # Elasticsearch / Kibana
export INFLUX_PASSWORD='rhksflwk1!'         # InfluxDB

# 앱 기동에 필요한 값 (7장·9장에서 발급받아 채운다)
export SECURITY_SECRET_KEY=''
export INFLUX_TOKEN=''

# 연합 트윈 연동용 카프카 주소 (연동할 카프카 서버의 주소를 넣는다. 예: 10.0.0.5:9092)
# 메타데이터 변경 내역을 twin-meta-changelog 토픽으로 보낸다
# 주소를 아직 못 받았으면 비워 두고 설치한 뒤 나중에 넣어도 된다
export KAFKA_SERVERS=''

export KAFKA_PASSWORD=''
EOF

# 비밀번호가 들어 있으므로 본인만 읽도록 권한을 제한한다
chmod 600 ~/dt.env

# 재접속할 때 자동으로 불러오도록 등록
grep -q 'source ~/dt.env' ~/.bashrc || echo 'source ~/dt.env' >> ~/.bashrc

# 현재 셸에 적용
source ~/dt.env

mkdir -p $DT_DIR

# 값 확인 - 브라우저에서 이 주소로 접속되는 게 맞는지 반드시 눈으로 확인한다

########################################
# 1-1. 기존 Docker 처리 방침
#
#      이미 도커가 설치된 서버라면 도커를 지우거나 바꾸지 않는다.
#      기존 이미지/컨테이너/볼륨을 그대로 두고, 부족한 것만 채운다.
#
#        docker 없음                        -> 1-2 실행 (설치)
#        docker 있고 docker compose 됨      -> 1-2, 1-3 건너뛰고 1-4 부터
#        docker 있고 docker compose 안 됨   -> 1-3 실행 (compose 만 추가)
#
#      ELK(5장) 설치에 docker compose v2 가 필요해서 이것만 맞추면 된다.
########################################


########################################
# 1-2. Docker 설치
#      도커가 설치되지 않은 서버에서만 실행한다.
########################################

# 설치 중 서비스 재시작 여부를 묻지 않도록 설정 (묻는 창에서 멈추는 것 방지)
sudo sed -i "s/^#\$nrconf{restart} = 'i';/\$nrconf{restart} = 'a';/" /etc/needrestart/needrestart.conf

# 패키지 목록 갱신
sudo apt-get update

# 도커 저장소 등록에 필요한 도구 설치
sudo apt-get install -y ca-certificates curl gnupg lsb-release

# 인증키를 넣을 디렉터리 생성
sudo install -m 0755 -d /etc/apt/keyrings

# 도커 공식 GPG 키를 내려받아 apt 형식으로 저장
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | \
  sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

# 모든 사용자가 키를 읽을 수 있도록 권한 부여
sudo chmod a+r /etc/apt/keyrings/docker.gpg

# 도커 공식 저장소를 apt 목록에 추가
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | \
  sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# 추가한 저장소를 반영
sudo apt-get update

# 도커 엔진과 compose/buildx 플러그인 설치
sudo NEEDRESTART_MODE=a apt-get install -y docker-ce docker-ce-cli containerd.io \
  docker-buildx-plugin docker-compose-plugin

# 설치된 버전 확인
sudo docker --version
sudo docker compose version

# 컨테이너가 실제로 실행되는지 확인 (Hello from Docker! 출력)
sudo docker run --rm hello-world


########################################
# 1-3. docker compose 만 추가  [조건부 - 1-2 를 실행했으면 건너뛴다]
#      도커는 있는데 docker compose 만 안 되는 서버에서 실행한다.
#      기존 도커는 그대로 두고 compose v2 플러그인 파일만 얹는다.
#      기존 docker-compose(v1) 도 지우지 않으므로 두 명령 다 사용할 수 있다.
########################################

# 플러그인을 넣을 디렉터리 생성
sudo mkdir -p /usr/local/lib/docker/cli-plugins

# 서버 아키텍처에 맞는 compose v2 실행 파일 내려받기
sudo curl -SL "https://github.com/docker/compose/releases/latest/download/docker-compose-linux-$(uname -m)" \
  -o /usr/local/lib/docker/cli-plugins/docker-compose

# 실행 권한 부여
sudo chmod +x /usr/local/lib/docker/cli-plugins/docker-compose

# 정상 인식되는지 확인
docker compose version

# 1-4. sudo 없이 docker 사용

# 재접속 없이 현재 셸에 바로 적용 (새 터미널부터는 자동 적용)
newgrp docker

# sudo 없이 되는지 확인
docker ps


########################################
# 1-5. 공용 도커 네트워크 생성
#      모든 컨테이너가 이 네트워크에서 컨테이너명으로 통신한다.
#      대역을 172.25.0.0/16 으로 고정해 게이트웨이를 항상 172.25.0.1 로 맞춘다.
#      (인증서버 설정의 dbHost 값이 서버마다 달라지지 않게 하기 위함)
########################################

# 172.25 대역이 이미 쓰이고 있는지 확인 (결과가 나오면 아래 대역을 다른 값으로 변경)
ip route | grep 172.25 || echo "172.25 대역 사용 안 함"

# keti-dt-net 네트워크 생성
docker network create --subnet 172.25.0.0/16 --gateway 172.25.0.1 keti-dt-net

# 생성 확인
docker network ls | grep keti-dt-net


########################################
# 1-6. 게이트웨이 IP 확인
#      컨테이너에서 호스트 PostgreSQL 로 접속할 때 쓰는 IP. 인증서버 설정에 사용한다.
########################################

# 172.25.0.1 로 나오는지 확인
docker network inspect keti-dt-net --format '{{(index .IPAM.Config 0).Gateway}}'


########################################
# 2-1. 기존 PostgreSQL 처리 방침
#
#      이미 설치된 PostgreSQL 이 있으면 새로 설치하거나 버전을 바꾸지 않는다.
#      기존 DB 를 그대로 두고 버전과 PostGIS 사용 가능 여부만 확인한다.
#
#        psql 없음            -> 2-2 실행 (설치)
#        psql 있고 12 이상    -> 2-2 건너뛰고 2-3 부터
#        psql 있고 12 미만    -> 서버 담당자와 협의 후 결정
########################################

# 설치 여부와 버전 확인

sudo -u postgres psql -c "select name, default_version from pg_available_extensions where name = 'postgis';"


########################################
# 2-2. PostgreSQL 16 + PostGIS 설치
#      DB 는 컨테이너가 아니라 서버에 직접 설치한다.
#      우분투 기본 저장소에는 14 만 있어서 PostgreSQL 공식 저장소를 추가해 16 을 설치한다.
########################################

# 저장소 등록에 필요한 도구 설치
sudo apt-get install -y curl ca-certificates gnupg lsb-release

# 인증키를 넣을 디렉터리 생성
sudo install -d /usr/share/postgresql-common/pgdg

# PostgreSQL 공식 GPG 키 저장
curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc | \

# 3-2. Virtuoso 계정 생성
#      여기서 만드는 meta_manager 계정이 Metadata API 의 접속 계정이다.
#      계정명은 환경변수 VIRTUOSO_USERNAME, 비밀번호는 VIRTUOSO_PASSWORD 에 대응한다.
#      기본값(meta_manager / rhksflwk1!)대로 만들면 환경변수를 넘기지 않아도 된다.
#
#      [방법 1] 아래 명령으로 만든다. 8890 을 열지 않아도 되고 브라우저도 필요 없다.
#      [방법 2] 8890 이 열려 있으면 Conductor 화면에서 만들어도 된다 (아래 주석 참고).
########################################

# 비밀번호에 ! 가 들어가면 bash 가 특수문자로 해석하므로 먼저 꺼둔다
set +H

# isql 을 짧게 쓰기 위한 함수. vdba 는 관리자로, vusr 은 만든 계정으로 실행한다
vdba() { docker exec virtuoso isql 1111 dba "$VIRTUOSO_DBA_PASSWORD" exec="$*"; }
vusr() { docker exec virtuoso isql 1111 "$VIRTUOSO_USERNAME" "$VIRTUOSO_PASSWORD" exec="$*"; }

# 계정 생성과 SPARQL 권한 부여
vdba "DB.DBA.USER_CREATE('$VIRTUOSO_USERNAME', '$VIRTUOSO_PASSWORD');"
vdba "DB.DBA.USER_GRANT_ROLE('$VIRTUOSO_USERNAME', 'SPARQL_SELECT', 0);"
vdba "DB.DBA.USER_GRANT_ROLE('$VIRTUOSO_USERNAME', 'SPARQL_UPDATE', 0);"
vdba "DB.DBA.USER_GRANT_ROLE('$VIRTUOSO_USERNAME', 'SPARQL_SPONGE', 0);"
vdba "DB.DBA.USER_GRANT_ROLE('$VIRTUOSO_USERNAME', 'SPARQL_LOAD_SERVICE_DATA', 0);"

# 기본 그룹을 dba 로 지정 (Conductor 의 Primary role : dba 에 해당)
# 이것을 하지 않으면 조회는 되지만 RDF 쓰기가 거부되어 10장에서 메타데이터 등록이 실패한다
vdba "DB.DBA.USER_SET_OPTION('$VIRTUOSO_USERNAME', 'PRIMARY_GROUP', 'dba');"

# 계정이 만들어졌는지 확인 (계정명이 나와야 한다)
vdba "select U_NAME from DB.DBA.SYS_USERS where U_NAME='$VIRTUOSO_USERNAME';"

# 조회 권한 확인 (결과가 나오면 정상)
vusr "SPARQL select * where { ?s ?p ?o } limit 5;"

# 쓰기 권한 확인 (Done. 이 나와야 한다. 앱이 메타데이터를 등록할 때 쓰는 경로다)
vusr "SPARQL INSERT DATA { GRAPH <http://test/g> { <http://test/s> <http://test/p> 'v' } };"

# 넣은 값이 조회되는지 확인
vusr "SPARQL SELECT * WHERE { GRAPH <http://test/g> { ?s ?p ?o } };"

# 확인용 데이터 삭제
vusr "SPARQL CLEAR GRAPH <http://test/g>;"


# [방법 2] Conductor 화면에서 만들 때 (8890 이 열려 있어야 한다)
#   echo "http://$SERVER_HOST:8890/conductor"
#
#   1) dba / VIRTUOSO_DBA_PASSWORD 로 로그인

#        User type     : SQL/ODBC

#        SPARQL_LOAD_SERVICE_DATA / SPARQL_SELECT / SPARQL_SPONGE / SPARQL_UPDATE
#   4) Default Permissions 전부 체크 후 저장
#   5) 만든 계정으로 재로그인 -> Linked Data - SPARQL 에서 select * where { ?s ?p ?o }

#   서버에 브라우저가 없고 8890 도 막혀 있으면 SSH 터널로 접근할 수 있다.

#   --env-file 은 컨테이너를 만들 때 한 번만 읽는다.
#     docker rm -f <컨테이너명> 후 docker run 을 다시 실행
#     docker exec <컨테이너명> printenv <변수명> 으로 반영 확인
#
#   값은 ~/dt.env 부터 고치고 앱 .env 를 다시 만든다.

# [막힐 때]
#   redirect_uri mismatch      -> DB 의 metaweb 앱 redirect_uri 와 웹 빌드값이 다름

#   로그인 화면 무한 반복      -> 7-2 에서 키를 바꾼 뒤 API 를 재시작하지 않음
#                                 docker restart keti-dt-metadata-api keti-dt-platformadmin-api
#   로그인 후 메뉴가 비어 있음 -> common.tb_com_authusermap 에 그 계정 권한이 없음
#   저장은 되는데 키워드 500   -> INFLUX_TOKEN 이 틀림 (아래로 확인)
#                                 docker logs keti-dt-metadata-api | grep -i unauthorized

# DB 에 들어 있는 redirect_uri 확인
